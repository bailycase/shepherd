"""Run only after final Release build and all competing workers/builds settle."""
from pathlib import Path
import json, os, subprocess, sys, time
root = Path('/Users/bailycase/Developer/Shepherd-perf-integration-2026-11-02')
out = Path('/tmp/shepherd-perf-fixes/final-benchmarks')
out.mkdir(parents=True, exist_ok=True)
release = root / '.build/arm64-apple-macosx/release'
archived_source = root / 'docs/audit/2026-11-02/artifacts/ExtraDataPaths.swift.txt'
source = out / 'ExtraDataPaths.swift'
source.write_bytes(archived_source.read_bytes())
objects = [str(p) for target in ('ShepherdCore','ShepherdProtocol','ShepherdRemote','ShepherdTestIsolation') for p in sorted((release / (target + '.build')).glob('*.o'))]
assert objects and source.is_file()
env = {k:v for k,v in os.environ.items() if not k.startswith(('SHEPHERD_', 'PI_'))}
env.update(PATH='/usr/bin:/bin:/usr/sbin:/sbin:/opt/homebrew/bin', TMPDIR='/tmp', LANG='en_US.UTF-8', GIT_CONFIG_GLOBAL='/dev/null', GIT_CONFIG_SYSTEM='/dev/null', GIT_CONFIG_NOSYSTEM='1', GIT_TERMINAL_PROMPT='0')

def run(name, command, timeout, additions=None):
    directory = out / name
    directory.mkdir(exist_ok=True)
    home = directory / 'home'
    home.mkdir(exist_ok=True)
    local = dict(env, HOME=str(home), **(additions or {}))
    (directory/'command.json').write_text(json.dumps({'command':command,'cwd':str(root),'timeout':timeout,'HOME':str(home),'additions':additions}, indent=2))
    start = time.monotonic()
    with (directory/'full.log').open('w') as log:
        result = subprocess.run(command, cwd=root, env=local, stdout=log, stderr=subprocess.STDOUT, timeout=timeout)
    (directory/'result.json').write_text(json.dumps({'exit':result.returncode,'elapsed':time.monotonic()-start},indent=2))
    if result.returncode:
        raise SystemExit(f'{name}: failed ({result.returncode}); inspect log')
    print(name, 'passed', flush=True)

(out/'revision.txt').write_text(subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True))
(out/'swift-version.txt').write_text(subprocess.check_output(['swift','--version'],text=True))
run('compile-extra', ['xcrun','swiftc','-O','-parse-as-library','-swift-version','5','-I',str(release/'Modules'),str(source),*objects,'-o',str(out/'extra-data-paths')],120)
for repetition in range(1,4):
    run(f'extra-{repetition}', [str(out/'extra-data-paths')],120)
    run(f'datapath-{repetition}', ['swift','test','--build-system','native','-c','release','--skip-build','--filter','DataPathBenchmarks'],600,{'SHEPHERD_BENCHMARK':'1','SHEPHERD_PERF_REPORT':'1'})
